cmake_minimum_required(VERSION 3.16)
project(mpc_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(mpc_spi main.cpp)
target_compile_options(mpc_spi PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(protocol_test protocol_test.cpp)
target_compile_options(protocol_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME protocol COMMAND protocol_test)

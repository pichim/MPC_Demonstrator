cmake_minimum_required(VERSION 3.16)
project(mpc_host LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "mpc_spi supports Linux only.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(mpc_spi src/main.cpp)
target_include_directories(mpc_spi PRIVATE lib/Spi)
target_compile_options(mpc_spi PRIVATE -Wall -Wextra -Wpedantic -Werror)

option(MPC_BUILD_CH347 "Build the laptop host for a CH347T USB-to-SPI adapter" OFF)
if(MPC_BUILD_CH347)
    set(MPC_CH347_SPI_SPEED_HZ 15000000 CACHE STRING "CH347T SPI clock in Hz")
    set(MPC_CH347_PERIOD_US 1000 CACHE STRING "CH347T host control period in microseconds")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
    add_executable(mpc_spi_usb src/main.cpp lib/Spi/spi_ch347.cpp)
    target_include_directories(mpc_spi_usb PRIVATE lib/Spi)
    target_compile_definitions(mpc_spi_usb PRIVATE MPC_USE_CH347
        SPI_SPEED_HZ=${MPC_CH347_SPI_SPEED_HZ} PERIOD_US=${MPC_CH347_PERIOD_US})
    target_compile_options(mpc_spi_usb PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(mpc_spi_usb PRIVATE PkgConfig::LIBUSB)
endif()
